    26→export function extractJsonArray(text: string): any[] | null {
    27→  if (!text) return null
    28→  // Strip markdown fences (```json ... ``` or ``` ... ```)
    29→  let cleaned = text.trim()
    30→  cleaned = cleaned.replace(/```(?:json)?/gi, '')
    31→  const first = cleaned.indexOf('[')
    32→  const last = cleaned.lastIndexOf(']')
    33→  if (first === -1 || last === -1 || last <= first) return null
    34→  try {
    35→    const parsed = JSON.parse(cleaned.slice(first, last + 1))
    36→    return Array.isArray(parsed) ? parsed : null
    37→  } catch {
    38→    return null
    39→  }
    40→}

    42→// ---------- 2. callLLM ----------
    43→export async function callLLM(system: string, user: string, retries = 2): Promise<string | null> {
    44→  for (let attempt = 0; attempt <= retries; attempt++) {
    45→    try {
    46→      const zai = await getZAI()
    47→      const completion = await zai.chat.completions.create({
    48→        messages: [
    49→          { role: 'system', content: system },
    50→          { role: 'user', content: user },
    51→        ],
    52→        thinking: { type: 'disabled' },
    53→      })
    54→      const raw = completion?.choices?.[0]?.message?.content
    55→      let text: string | null = null
    56→      if (typeof raw === 'string') {
    57→        text = raw
    58→      } else if (Array.isArray(raw)) {
    59→        text = raw.map((part: any) => (typeof part === 'string' ? part : part?.text ?? '')).join('')
    60→      }
    61→      if (text && text.trim().length > 0) return text
    62→      console.warn(`[callLLM] attempt ${attempt + 1}: empty content`)
    63→    } catch (err) {
    64→      console.error(`[callLLM] attempt ${attempt + 1} error:`, err)
    65→    }
    66→    if (attempt < retries) {
    67→      await new Promise((resolve) => setTimeout(resolve, 1000))
    68→    }
    69→  }
    70→  console.error('[callLLM] final failure after retries')
    71→  return null
    72→}

    80→function questionTypeFlags(questionType: string) {
    81→  const t = (questionType || '').toLowerCase()
    82→  return {
    83→    isPG: t.includes('pilihan ganda'),
    84→    isComplex: t.includes('kompleks'),
    85→    isBS: t.includes('benar') || t.includes('salah'),
    86→    isShort: t.includes('isian'),
    87→    isEssay: t.includes('uraian'),
    88→  }
    89→}

    91→function stripOptionPrefix(opt: string): string {
    92→  let s = String(opt ?? '').trim()
    93→  const stripped = s.replace(/^[A-Ja-j][.)]\s+/, '')
    94→  if (stripped.trim().length > 0) s = stripped
    95→  return s
    96→}

    98→function normalizeAnswerLetter(answer: any, optionCount: number): string | null {
    99→  const raw = String(answer ?? '').trim().toUpperCase()
   100→  if (!raw) return null
   101→  const firstLetter = raw.match(/[A-J]/)
   102→  if (!firstLetter) return null
   103→  const letter = firstLetter[0]
   104→  const idx = LETTERS.indexOf(letter)
   105→  if (idx === -1 || idx >= optionCount) return null
   106→  return letter
   107→}
   108→
   109→function normalizeAnswerLetters(answer: any, optionCount: number): string | null {
   110→  const raw = String(answer ?? '').trim().toUpperCase()
   111→  if (!raw) return null
   112→  const letters = raw.match(/[A-J]/g) ?? []
   113→  const valid = Array.from(new Set(letters)).filter((l) => LETTERS.indexOf(l) < optionCount)
   114→  if (valid.length === 0) return null
   115→  return valid.join(',')
   116→}

   118→// ---------- Difficulty & cognitive helpers ----------
   119→function difficultySequence(structure: StructureData): string[] {
   120→  const seq: string[] = []
   121→  for (let i = 0; i < structure.easyCount; i++) seq.push('Mudah')
   122→  for (let i = 0; i < structure.mediumCount; i++) seq.push('Sedang')
   123→  for (let i = 0; i < structure.hardCount; i++) seq.push('Sulit')
   124→  const total = structure.total > 0 ? structure.total : seq.length
   125→  while (seq.length < total) seq.push('Sedang')
   126→  return seq.slice(0, total)
   127→}

   129→function cognitiveAt(structure: StructureData, index: number): string {
   130→  const levels = structure.cognitiveLevels && structure.cognitiveLevels.length > 0 ? structure.cognitiveLevels : ['C1']
   131→  return levels[index % levels.length]
   132→}

   134→// ---------- 4. fallbackQuestions ----------
   135→export function fallbackQuestionAt(
   136→  profile: ExamProfileData,
   137→  structure: StructureData,
   138→  index: number,
   139→  difficulty: string,
   140→  cognitiveLevel: string,
   141→  number: number
   142→): NewQuestion {
   143→  const { isPG, isComplex, isBS, isShort, isEssay } = questionTypeFlags(structure.questionType)
   144→  const isMath = /matematika/i.test(profile.subject || '')
   145→  const indicator = `Siswa mampu menyelesaikan soal tentang ${profile.topic}`
   146→  const material = profile.topic
   147→  const source = 'Buku Siswa Kurikulum Merdeka'
   148→  const base = { number, type: structure.questionType, difficulty, cognitiveLevel, indicator, material, source, scoreWeight: 1, imagePrompt: null, imagePath: null }
   149→
   150→  if (isMath) {
   151→    const a = 2 + ((index * 7) % 8)
   152→    const b = 1 + ((index * 5) % 9)
   153→    const usePlus = index % 2 === 0
   154→    let aa = a
   155→    let bb = b
   156→    if (!usePlus && bb > aa) {
   157→      const tmp = aa
   158→      aa = bb
   159→      bb = tmp
   160→    }
   161→    const op = usePlus ? '+' : '-'
   162→    const correct = usePlus ? aa + bb : aa - bb
   163→    const story = index % 3 !== 0
   164→    const items = ['apel', 'buku', 'kelereng']
   165→    const names = ['Andi', 'Budi', 'Citra']
   166→    const item = items[index % 3]
   167→    const name = names[index % 3]
   168→    const storyContent = `${name} memiliki ${aa} ${item}. Kemudian ${usePlus ? `ditambah ${bb}` : `dikurangi ${bb}`}. Berapa hasil akhirnya?`
   169→    const plainContent = `Hasil dari ${aa} ${op} ${bb} adalah…`
   170→
   171→    if (isBS) {
   172→      const claimed = correct + (index % 2 === 0 ? 0 : 1 + (index % 3))
   173→      const content = `Perhatikan pernyataan berikut: "Hasil dari ${aa} ${op} ${bb} adalah ${claimed}." Pernyataan tersebut benar atau salah?`
   174→      return { ...base, content, options: ['Benar', 'Salah'], answer: claimed === correct ? 'A' : 'B' }
   175→    }
   176→
   177→    if (isShort || isEssay) {
   178→      const content = isEssay ? storyContent : plainContent
   179→      const answer = isEssay
   180→        ? `${correct}, karena ${aa} ${usePlus ? 'ditambah' : 'dikurangi'} ${bb} menghasilkan ${correct}.`
   181→        : String(correct)
   182→      return { ...base, content, options: [], answer }
   183→    }
   184→
   185→    // PG / PG Kompleks
   186→    const content = story ? storyContent : plainContent
   187→    const correctPos = Math.min(index % 3, Math.max(structure.optionCount - 1, 0))
   188→    const deltas = [-1, 1, 2, -2, 3, -3, 4]
   189→    const distractors: string[] = []
   190→    for (const d of deltas) {
   191→      if (distractors.length >= structure.optionCount - 1) break
   192→      const v = correct + d
   193→      if (v >= 0 && v !== correct && !distractors.includes(String(v))) distractors.push(String(v))
   194→    }
   195→    const options: string[] = []
   196→    let dIdx = 0
   197→    for (let k = 0; k < structure.optionCount; k++) {
   198→      if (k === correctPos) {
   199→        options.push(String(correct))
   200→      } else {
   201→        options.push(distractors[dIdx % distractors.length])
   202→        dIdx++
   203→      }
   204→    }
   205→    const answer = isComplex ? LETTERS[correctPos] : LETTERS[correctPos]
   206→    return { ...base, content, options, answer }
   207→  }
   208→
   209→  // Generic (non-matematika)
   210→  if (isBS) {
   211→    const positive = index % 2 === 0
   212→    const content = positive
   213→      ? `Pernyataan: "Konsep ${profile.topic} termasuk materi penting yang dipelajari dalam ${profile.subject} di ${profile.grade}." Pernyataan tersebut benar atau salah?`
   214→      : `Pernyataan: "${profile.topic} sama sekali tidak memiliki kaitan dengan materi ${profile.subject}." Pernyataan tersebut benar atau salah?`
   215→    return { ...base, content, options: ['Benar', 'Salah'], answer: positive ? 'A' : 'B' }
   216→  }
   217→
   218→  if (isShort || isEssay) {
   219→    const content = `Jelaskan dengan singkat konsep ${profile.topic} (${index + 1})${isEssay ? ' beserta satu contohnya' : ''}.`
   220→    const answer = isEssay
   221→      ? `Jawaban model: siswa menjelaskan pengertian ${profile.topic} dan memberikan satu contoh relevan dalam kehidupan sehari-hari.`
   222→      : `${profile.topic} merupakan konsep dalam ${profile.subject} yang mencakup pemahaman, penerapan, dan contoh nyata di kehidupan sehari-hari.`
   223→    return { ...base, content, options: [], answer }
   224→  }
   225→
   226→  // PG / PG Kompleks
   227→  const genericOptions = ['Pernyataan yang tepat', 'Pernyataan kurang tepat', 'Tidak relevan']
   228→  while (genericOptions.length < structure.optionCount) {
   229→    genericOptions.push(`Pernyataan tidak dapat diterima (${genericOptions.length})`)
   230→  }
   231→  return {
   232→    ...base,
   233→    content: `Pilih pernyataan yang paling tepat mengenai konsep ${profile.topic} (soal ${index + 1}).`,
   234→    options: genericOptions.slice(0, structure.optionCount),
   235→    answer: 'A',
   236→  }
   237→}

   239→export function fallbackQuestions(profile: ExamProfileData, structure: StructureData, startNumber: number): NewQuestion[] {
   240→  const difficulties = difficultySequence(structure)
   241→  const total = structure.total > 0 ? structure.total : difficulties.length
   242→  const out: NewQuestion[] = []
   243→  for (let i = 0; i < total; i++) {
   244→    out.push(fallbackQuestionAt(profile, structure, i, difficulties[i], cognitiveAt(structure, i), startNumber + i))
   245→  }
   246→  return out
   247→}

   249→// ---------- 3. generateQuestionsForStructure ----------
   250→function sanitizeLLMQuestion(
   251→  raw: any,
   252→  profile: ExamProfileData,
   253→  structure: StructureData,
   254→  difficulty: string,
   255→  cognitiveLevel: string,
   256→  number: number
   257→): NewQuestion | null {
   258→  const { isPG, isComplex, isBS, isShort, isEssay } = questionTypeFlags(structure.questionType)
   259→  const content = String(raw?.content ?? '').trim()
   260→  if (!content) return null
   261→
   262→  let options: string[] = []
   263→  let answer = ''
   264→
   265→  if (isBS) {
   266→    options = ['Benar', 'Salah']
   267→    const ans = String(raw?.answer ?? '').trim().toLowerCase()
   268→    if (ans.includes('benar') || (ans.startsWith('a') && !ans.includes('benar') && !ans.includes('salah'))) answer = 'A'
   269→    else if (ans.includes('salah') || ans.startsWith('b')) answer = 'B'
   270→    else return null
   271→  } else if (isPG) {
   272→    if (!Array.isArray(raw?.options)) return null
   273→    options = raw.options.map(stripOptionPrefix).filter((o: string) => o.length > 0)
   274→    if (options.length > structure.optionCount) options = options.slice(0, structure.optionCount)
   275→    if (options.length !== structure.optionCount) return null
   276→    const normalized = isComplex
   277→      ? normalizeAnswerLetters(raw?.answer, structure.optionCount)
   278→      : normalizeAnswerLetter(raw?.answer, structure.optionCount)
   279→    if (!normalized) return null
   280→    answer = normalized
   281→  } else if (isShort || isEssay) {
   282→    options = []
   283→    answer = String(raw?.answer ?? '').trim()
   284→    if (!answer) return null
   285→  } else {
   286→    // Unknown type: treat as short answer
   287→    options = []
   288→    answer = String(raw?.answer ?? '').trim()
   289→    if (!answer) return null
   290→  }
   291→
   292→  const indicator = String(raw?.indicator ?? '').trim() || `Siswa mampu menyelesaikan soal tentang ${profile.topic}`
   293→  const material = String(raw?.material ?? '').trim() || profile.topic
   294→  const source = String(raw?.source ?? '').trim() || 'Buku Siswa Kurikulum Merdeka'
   295→
   296→  return {
   297→    number,
   298→    type: structure.questionType,
   299→    difficulty,
   300→    cognitiveLevel,
   301→    content,
   302→    options,
   303→    answer,
   304→    indicator,
   305→    material,
   306→    source,
   307→    scoreWeight: 1,
   308→    imagePrompt: null,
   309→    imagePath: null,
   310→  }
   311→}

   313→export async function generateQuestionsForStructure(
   314→  profile: ExamProfileData,
   315→  structure: StructureData,
   316→  startNumber: number
   317→): Promise<NewQuestion[]> {
   318→  const difficulties = difficultySequence(structure)
   319→  const total = structure.total > 0 ? structure.total : difficulties.length
   320→  const CHUNK_SIZE = 6
   321→  const out: NewQuestion[] = []
   322→
   323→  for (let chunkStart = 0; chunkStart < total; chunkStart += CHUNK_SIZE) {
   324→    const chunkEnd = Math.min(chunkStart + CHUNK_SIZE, total)
   325→    const chunkIndexes: number[] = []
   326→    for (let i = chunkStart; i < chunkEnd; i++) chunkIndexes.push(i)
   327→
   328→    const levels = structure.cognitiveLevels && structure.cognitiveLevels.length > 0 ? structure.cognitiveLevels.join(', ') : 'C1'
   329→    const chunkDifficulties = chunkIndexes.map((i) => difficulties[i]).join(', ')
   330→
   331→    const userPrompt = [
   332→      `Buat soal ujian dengan konteks berikut: jenjang ${profile.level}, fase ${profile.phase}, kelas ${profile.grade}, mata pelajaran ${profile.subject}, topik "${profile.topic}".`,
   333→      `Bentuk soal: ${structure.questionType} dengan ${structure.optionCount} opsi (untuk Pilihan Ganda / Pilihan Ganda Kompleks).`,
   334→      structure.customInstruction ? `Instruksi tambahan dari guru: ${structure.customInstruction}` : '',
   335→      `Buat ${chunkIndexes.length} soal dengan spesifikasi per soal (array JSON objects): {"content": string (teks soal jelas sesuai tingkat kelas), "options": array ${structure.optionCount} string TANPA huruf prefix (HANYA untuk bentuk pilihan ganda; untuk "Benar/Salah" gunakan ["Benar","Salah"]; untuk "Isian Singkat"/"Uraian" gunakan []), "answer": huruf kunci (A/B/C/...) untuk pilihan ganda, atau jawaban model singkat untuk isian/uraian, atau "A" (Benar) / "B" (Salah) untuk Benar/Salah, "indicator": string indikator soal, "material": string materi pokok, "source": string buku sumber/acuan}.`,
   336→      `Tingkat kesulitan untuk ${chunkIndexes.length} soal ini (berurutan): ${chunkDifficulties}.`,
   337→      `Soal harus variatif, kontekstual, sesuai kognitif ${levels}, dan tidak saling mirip. Balas HANYA JSON array.`,
   338→    ]
   339→      .filter(Boolean)
   340→      .join('\n')
   341→
   342→    console.log(`[generate] chunk ${chunkStart}-${chunkEnd - 1} (${chunkIndexes.length} soal) dari ${total} — memanggil LLM...`)
   343→    const text = await callLLM(SYSTEM_PROMPT, userPrompt, 2)
   344→    const parsed = text ? extractJsonArray(text) : null
   345→
   346→    if (!parsed) {
   347→      console.warn(`[generate] chunk ${chunkStart}-${chunkEnd - 1}: LLM gagal/parse gagal → menggunakan fallback template`)
   348→    }
   349→
   350→    for (let k = 0; k < chunkIndexes.length; k++) {
   351→      const i = chunkIndexes[k]
   352→      const number = startNumber + i
   353→      const difficulty = difficulties[i]
   354→      const cognitiveLevel = cognitiveAt(structure, i)
   355→      let question: NewQuestion | null = null
   356→      if (parsed && parsed[k]) {
   357→        question = sanitizeLLMQuestion(parsed[k], profile, structure, difficulty, cognitiveLevel, number)
   358→      }
   359→      if (!question) {
   360→        question = fallbackQuestionAt(profile, structure, i, difficulty, cognitiveLevel, number)
   361→      }
   362→      out.push(question)
   363→    }
   364→    console.log(`[generate] chunk ${chunkStart}-${chunkEnd - 1} selesai (total soal sejauh ini: ${out.length})`)
   365→  }
   366→
   367→  return out
   368→}

   370→// ---------- 5. regenerateQuestion ----------
   371→export async function regenerateQuestion(
   372→  profile: ExamProfileData,
   373→  structure: StructureData,
   374→  question: QuestionData
   375→): Promise<{ content: string; options: string[]; answer: string; indicator: string; material: string; source: string }> {
   376→  const { isPG, isComplex, isBS, isShort, isEssay } = questionTypeFlags(structure.questionType)
   377→  const levels = structure.cognitiveLevels && structure.cognitiveLevels.length > 0 ? structure.cognitiveLevels.join(', ') : 'C1'
   378→
   379→  const optionRule = isBS
   380→    ? '"options": ["Benar","Salah"], "answer": "A" (Benar) atau "B" (Salah)'
   381→    : isPG
   382→      ? `"options": array ${structure.optionCount} string TANPA huruf prefix, "answer": huruf kunci (${isComplex ? 'boleh beberapa huruf dipisah koma' : 'satu huruf A/B/C/...'})`
   383→      : isShort || isEssay
   384→        ? '"options": [], "answer": jawaban model singkat'
   385→        : '"options": [], "answer": jawaban model singkat'
   386→
   387→  const userPrompt = [
   388→    `Buat SOAL BARU yang berbeda dari soal sebelumnya berikut ini (jangan menyalin isi soal lama).`,
   389→    `Soal lama: "${question.content.slice(0, 400)}"`,
   390→    `Konteks: jenjang ${profile.level}, fase ${profile.phase}, kelas ${profile.grade}, mata pelajaran ${profile.subject}, topik "${profile.topic}".`,
   391→    `Bentuk soal: ${structure.questionType} dengan ${structure.optionCount} opsi (untuk Pilihan Ganda / Pilihan Ganda Kompleks). Tingkat kesulitan: ${question.difficulty}. Level kognitif: ${question.cognitiveLevel} dari ${levels}.`,
   392→    structure.customInstruction ? `Instruksi tambahan dari guru: ${structure.customInstruction}` : '',
   393→    `Balas dengan JSON array yang berisi SATU object soal dengan spesifikasi: {"content": string, ${optionRule}, "indicator": string indikator soal, "material": string materi pokok, "source": string buku sumber/acuan}. Soal harus variatif dan kontekstual. Balas HANYA JSON array.`,
   394→  ]
   395→    .filter(Boolean)
   396→    .join('\n')
   397→
   398→  console.log(`[regenerate] soal nomor ${question.number} — memanggil LLM...`)
   399→  const text = await callLLM(SYSTEM_PROMPT, userPrompt, 2)
   400→  const parsed = text ? extractJsonArray(text) : null
   401→  const raw = parsed && parsed.length > 0 ? parsed[0] : null
   402→
   403→  if (raw) {
   404→    const sanitized = sanitizeLLMQuestion(
   405→      raw,
   406→      profile,
   407→      { ...structure, total: 1 },
   408→      question.difficulty,
   409→      question.cognitiveLevel,
   410→      question.number
   411→    )
   412→    if (sanitized) {
   413→      console.log(`[regenerate] soal nomor ${question.number}: LLM OK`)
   414→      return {
   415→        content: sanitized.content,
   416→        options: sanitized.options,
   417→        answer: sanitized.answer,
   418→        indicator: sanitized.indicator,
   419→        material: sanitized.material,
   420→        source: sanitized.source,
   421→      }
   422→    }
   423→  }
   424→
   425→  console.warn(`[regenerate] soal nomor ${question.number}: LLM gagal/tidak valid → fallback template variant`)
   426→  const variant = fallbackQuestionAt(
   427→    profile,
   428→    structure,
   429→    question.number * 2 + 1,
   430→    question.difficulty,
   431→    question.cognitiveLevel,
   432→    question.number
   433→  )
   434→  return {
   435→    content: variant.content,
   436→    options: variant.options,
   437→    answer: variant.answer,
   438→    indicator: variant.indicator,
   439→    material: variant.material,
   440→    source: variant.source,
   441→  }
   442→}

   557→function rotateTkaSet(set: TkaFallbackSet, variant: number): TkaQuestionData {
   558→  const n = set.options.length
   559→  const s = ((variant % n) + n) % n
   560→  const rotated = set.options.map((_, k) => set.options[(k - s + n) % n])
   561→  const answerIdx = LETTERS.indexOf(set.answer)
   562→  const newAnswerIdx = (answerIdx + s) % n
   563→  const leads = ['Berdasarkan stimulus tersebut, ', 'Berdasarkan teks di atas, ', 'Berdasarkan bacaan tersebut, ']
   564→  let content = set.content
   565→  if (variant > 0 && !content.toLowerCase().startsWith('berdasarkan')) {
   566→    content = leads[variant % leads.length] + content.charAt(0).toLowerCase() + content.slice(1)
   567→  }
   568→  return {
   569→    stimulus: set.stimulus,
   570→    content,
   571→    options: rotated,
   572→    answer: LETTERS[newAnswerIdx],
   573→    cognitiveLevel: set.cognitiveLevel,
   574→    difficulty: set.difficulty,
   575→  }
   576→}

   578→export function fallbackTkaQuestionAt(subtest: string, index: number): TkaQuestionData {
   579→  const isLiterasi = /literasi|bahasa/i.test(subtest || '')
   580→  const isNumerasi = /numerasi|matematika/i.test(subtest || '')
   581→  const pool = isLiterasi ? [0, 1, 2] : isNumerasi ? [3, 4, 5] : [0, 1, 2, 3, 4, 5]
   582→  const setIdx = pool[index % pool.length]
   583→  const variant = Math.floor(index / pool.length)
   584→  return rotateTkaSet(TKA_FALLBACK_SETS[setIdx], variant)
   585→}

   587→export async function generateTkaQuestions(level: string, subtest: string, count: number): Promise<TkaQuestionData[]> {
   588→  const CHUNK_SIZE = 5
   589→  const out: TkaQuestionData[] = []
   590→
   591→  for (let chunkStart = 0; chunkStart < count; chunkStart += CHUNK_SIZE) {
   592→    const chunkCount = Math.min(CHUNK_SIZE, count - chunkStart)
   593→    const theme = TKA_THEMES[(chunkStart / CHUNK_SIZE) % TKA_THEMES.length]
   594→
   595→    const userPrompt = [
   596→      `Buat ${chunkCount} soal standar Tes Kemampuan Akademik (TKA) Kemendikdasmen 2026 untuk jenjang ${level}, subtes ${subtest}.`,
   597→      `Soal harus HOTS berbasis stimulus dengan tema konteks batch ini: ${theme}.`,
   598→      `Setiap item adalah JSON object dengan spesifikasi: {"stimulus": string (120–200 kata; untuk Literasi Bahasa Indonesia: teks narasi/informasi menarik berita/tokoh/lingkungan/budaya Indonesia; untuk Numerasi/Matematika: konteks data/tabel naratif kehidupan nyata), "content": pertanyaan HOTS, "options": 4 opsi tanpa huruf, "answer": huruf kunci (A-D), "cognitiveLevel": "C4"|"C5"|"C6", "difficulty": "Sedang"|"Sulit"}.`,
   599→      `Soal harus variatif dan tidak saling mirip. Balas HANYA JSON array.`,
   600→    ].join('\n')
   601→
   602→    console.log(`[tka/generate] chunk ${chunkStart}-${chunkStart + chunkCount - 1} (${chunkCount} soal) tema: ${theme} — memanggil LLM...`)
   603→    const text = await callLLM(TKA_SYSTEM_PROMPT, userPrompt, 2)
   604→    const parsed = text ? extractJsonArray(text) : null
   605→
   606→    if (!parsed) {
   607→      console.warn(`[tka/generate] chunk ${chunkStart}: LLM gagal/parse gagal → fallback template`)
   608→    }
   609→
   610→    for (let k = 0; k < chunkCount; k++) {
   611→      const index = chunkStart + k
   612→      let item: TkaQuestionData | null = null
   613→      const raw = parsed && parsed[k] ? parsed[k] : null
   614→      if (raw) {
   615→        const stimulus = String(raw?.stimulus ?? '').trim()
   616→        const content = String(raw?.content ?? '').trim()
   617→        const options = Array.isArray(raw?.options) ? raw.options.map(stripOptionPrefix).filter((o: string) => o.length > 0) : []
   618→        const answer = normalizeAnswerLetter(raw?.answer, 4)
   619→        const cognitiveLevel = ['C4', 'C5', 'C6'].includes(String(raw?.cognitiveLevel)) ? String(raw.cognitiveLevel) : 'C4'
   620→        const difficulty = ['Sedang', 'Sulit'].includes(String(raw?.difficulty)) ? String(raw.difficulty) : 'Sedang'
   621→        if (stimulus.length >= 30 && content && options.length === 4 && answer) {
   622→          item = { stimulus, content, options, answer, cognitiveLevel, difficulty }
   623→        }
   624→      }
   625→      if (!item) {
   626→        item = fallbackTkaQuestionAt(subtest, index)
   627→      }
   628→      out.push(item)
   629→    }
   630→    console.log(`[tka/generate] chunk ${chunkStart} selesai (total soal sejauh ini: ${out.length})`)
   631→  }
   632→
   633→  return out
   634→}
